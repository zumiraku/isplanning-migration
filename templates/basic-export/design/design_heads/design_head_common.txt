<script>
//ページトップに戻る
$(function() {
     // トップに戻るボタン追従型＋最終位置指定
    $(function () {
      var footerInfo = $('.pageTop');
      footerInfo.hide();
      $(window).scroll(function () {
        if ($(this).scrollTop() > 300) {
          footerInfo.fadeIn();
        } else {
          footerInfo.fadeOut();
        }
      });
      footerInfo.click(function () {
        $('body,html').animate({
          scrollTop: 0
        }, 500);
        return false;
      });
    });
	
});
</script>
